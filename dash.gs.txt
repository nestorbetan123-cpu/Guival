
/* ---------- Dashboard (fase 4, 20/11/2026) ---------- */
/** Sube a Supabase lo del Dashboard: dims, quién ve qué copia y, por cada mes que cambió, una copia filtrada por acceso.
 *  Lo llama actualizarTablero_ con cada cierre; sbDashTodo lo hace con todos los meses (desde el editor). */
function sbDash_(todo, ms) {
  var dimsTxt = dimsTexto_(), dims = JSON.parse(dimsTxt);
  sbRobot_('robot_dash', { tipo: 'dims', datos: { dims: dimsTxt } });

  // Quién ve qué: una llave por cada forma distinta de acceso.
  var acc = accesos_(), llaves = {}, filas = [], como = {};
  try { appComoLista_().forEach(function (x) { como[x.clave] = true; }); } catch (e) {}
  Object.keys(acc).forEach(function (k) {
    var a = acceso_(k); if (!a) return;
    var ll = llaveAcceso_(a);
    llaves[ll] = a;
    filas.push([k, ll, a.tipo, a.etiqueta || '', !!como[k]]);
  });
  sbRobot_('robot_dash', { tipo: 'llaves', datos: filas });

  // Meses: solo los que cambiaron desde la última subida (o todos).
  var props = PropertiesService.getScriptProperties();
  var hechos = todo ? {} : JSON.parse(props.getProperty('SB_DASH_VER') || '{}');
  var firma = Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, JSON.stringify(filas))).slice(0, 10);
  var t0 = Date.now(), n = 0;
  (dims.meses || []).slice().reverse().forEach(function (mes) {   // el mes actual primero
    var ver = (dims.parts && dims.parts[mes]) || '';
    if (hechos[mes] === ver + '|' + firma) return;
    if (Date.now() - t0 > (ms || 4.5 * 60 * 1000)) return;            // lo que falte, en la siguiente pasada
    var partes = {};
    Object.keys(llaves).forEach(function (ll) { partes[ll] = parteParaAcc_(llaves[ll], mes); });
    sbRobot_('robot_dash', { tipo: 'parte', datos: { mes: mes, ver: ver, partes: partes } });
    hechos[mes] = ver + '|' + firma; n++;
    props.setProperty('SB_DASH_VER', JSON.stringify(hechos));
  });
  var faltan = (dims.meses || []).filter(function (m) { return hechos[m] !== ((dims.parts && dims.parts[m]) || '') + '|' + firma; }).length;
  return { meses: n, faltan: faltan, llaves: Object.keys(llaves).length };
}
/** Ejecutar desde el editor (repetir hasta que diga faltan: 0). */
function sbDashTodo() { Logger.log('Dashboard → Supabase: ' + JSON.stringify(sbDash_(false))); }
